. Prerequisites

sudo apt-get install gcc dpkg-dev gpg
package name scheme:
<package-name>_<version>-<release-number>_<architecture>

. Create a custom package

mkdir -pv hello-world_0.0.1-1_amd64/usr/bin/
mkdir -pv hello-world_0.0.1-1_amd64/DEBIAN/
cp hello_world hello-world_0.0.1-1_amd64/usr/bin
cp -v control hello-world_0.0.1-1_amd64/DEBIAN/

.. Creates the DEB file

$ dpkg --build hello-world_0.0.1-1_amd64
dpkg-deb: building package 'hello-world' in 'hello-world_0.0.1-1_amd64.deb'.

# Verifies the content
dpkg -c hello-world_0.0.1-1_amd64.deb

# Dumps the package information.
dpkg --info  hello-world_0.0.1-1_amd64.deb

.. Creates the local DEB file

# Installs the DEB package
sudo dpkg -i hello-world_0.0.1-1_amd64.deb

# Removes the DEB package
sudo apt-get remove hello-world

. Creates the apt repository

mkdir -pv sentilab-repo/pool/main/gateway-scanner/
mkdir -pv sentilab-repo/pool/main/gateway-objmanager/
mkdir -pv sentilab-repo/dists/stable/main/binary-amd64

cp -v hello-world_0.0.1-1_amd64.deb sentilab-repo/pool/main/gateway-scanner/

( cd sentilab-repo && dpkg-scanpackages --arch amd64 pool/ )
( cd sentilab-repo && \
  dpkg-scanpackages --arch amd64 pool/ > dists/stable/main/binary-amd64/Packages && \
  cat dists/stable/main/binary-amd64/Packages && \
  cat dists/stable/main/binary-amd64/Packages | gzip -9 > dists/stable/main/binary-amd64/Packages.gz && \
  ls -lah dists/stable/main/binary-amd64
)

.. Generate the release file

( cd sentilab-repo/dists/stable &&
  ../../../generate-release.sh > Release &&
  cat Release )

.. Test the repo with a HTTP server

  python3 -m http.server

.. Add the repo to the apt repositories

  sudo apt-get update --allow-insecure-repositories
  sudo apt-get install hello-world

  # Disinstall the package and remove the repo from apt.
  sudo apt-get remove hello-world
  sudo rm -v /etc/apt/sources.list.d/sentilab.list
  sudo apt-get update

. Sign the repository

.. Generate the key-pair

  rm -rfv pgp-keys-sentilab
  mkdir -vp pgp-keys-sentilab

  export GNUPGHOME="$(pwd)/pgp-keys-sentilab"
  echo "GNUPGHOME: ${GNUPGHOME}"
  gpgconf --kill gpg-agent # Restart the GPG agent

  gpg -v --no-tty --batch --gen-key generate-sentilab-pgp-key.batch

  gpg --list-keys

  # Exports a single key
  gpg --armor --export sentilab > sentilab-pgp-key.public

  # List the packets of an exported key, there should be only one.
  gpg --armor --export sentilab | gpg --list-packets

  # Exports the sentilab private key.
  gpg --armor --export-secret-keys sentilab > sentilab-pgp-key.private

.. Signes the Release file

  rm -rfv pgp-keys-sentilab
  mkdir -vp pgp-keys-sentilab-repo
  export GNUPGHOME="$(pwd)/pgp-keys-sentilab-repo"
  echo "GNUPGHOME: ${GNUPGHOME}"
  gpgconf --kill gpg-agent # Restart the GPG agent

  gpg --list-keys # Should be empty

  # Imports the Sentilab secret key in the Sentilab Repo PGP key store.
  cat sentilab-pgp-key.private | gpg -v --import

  gpg --list-keys # There should be one secret key.

  cat sentilab-repo/dists/stable/Release | gpg --default-key sentilab -abs > sentilab-repo/dists/stable/Release.gpg
  ls -la sentilab-repo/dists/stable

  # Combines Release and Release.pgp
  cat sentilab-repo/dists/stable/Release | gpg --default-key sentilab -abs --clearsign > sentilab-repo/dists/stable/InRelease
